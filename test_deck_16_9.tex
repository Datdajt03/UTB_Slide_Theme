\subsection{Môi trường định nghĩa \& khối ghi chú}
\begin{frame}{Môi trường định nghĩa và khối ghi chú}{Đồng bộ màu sắc nhận diện và chú thích tự động}
  \begin{definition}[Phép toán hai ngôi]
    Cho tập $X \neq \varnothing$. Một \note{phép toán hai ngôi}{Một ánh xạ từ $X^2 \to X$ gán mỗi cặp phần tử $(x, y)$ thành một phần tử $x * y \in X$.} trên tập $X$, ký hiệu là $*$, là một quy tắc biến đổi mỗi cặp phần tử $(x, y) \in X^2$ thành phần tử $x * y \in X$. Nói cách khác, phép toán $*$ là một ánh xạ:
    \[
      (*) : X^2 \longrightarrow X, \quad (x, y) \longmapsto x * y
    \]
  \end{definition}

  \begin{exampleblock}{Nhận xét quan trọng}
    \warning Các môi trường khối và toán học chuẩn của Beamer (\texttt{block}, \texttt{definition}, \texttt{theorem}, \texttt{exampleblock}) được tinh chỉnh tự động theo bảng màu thương hiệu UTB, hỗ trợ đánh số và tham chiếu chéo thông minh qua \texttt{cleveref}.

\subsection{Tùy biến danh sách (enumitem \& tasks)}
\begin{frame}{Tùy biến danh sách với enumitem và tasks}{Đánh số dạng ô vuông nhận diện thương hiệu UTB}
  \begin{definition}[Nghiệm Pareto]
    \begin{utbsqlist}
      \item $x^* \in C$ là \emph{nghiệm Pareto} nếu $\nexists x \in C$ sao cho $g_i(x) \le g_i(x^*)$ với mọi $i$ và $g_j(x) < g_j(x^*)$ tại ít nhất một chỉ số $j$. \label{def:pareto}
      \item $\bar{x} \in C$ là \emph{nghiệm Pareto yếu} nếu $\nexists x \in C$ sao cho $g_i(x) < g_i(\bar{x})$ với mọi $i$.
      \item Tập hợp tất cả các nghiệm Pareto được gọi là \emph{tập Pareto}. Ảnh của nó trong không gian mục tiêu được gọi là \emph{biên Pareto}.
    \end{utbsqlist}
  \end{definition}
  \vskip 2pt
  \textbf{Phân loại bài toán tối ưu với \texttt{tasks}} (gọi tùy biến với badge ô vuông chữ cái):

        \caption{Thuật toán MPG-E}\label{algo:mpg}
        \KwIn{$x_0 \in \mathcal{X}$, $\beta \in (0,1)$, $\gamma > 0$}
        \KwOut{Nghiệm Pareto yếu $x^*$}
        \BlankLine
        \For{$k = 0, 1, 2, \dots$}{
          $d_k = \argmin_{d} \left\{ \max_i \innerp{\nabla f_i(x_k)}{d} + \frac{1}{2}\norm{d}^2 \right\}$\;
          \If{$\norm{d_k} < \epsilon$}{
            \Return $x_k$\;
          }
          $x_{k+1} = \mathrm{Proj}_{\mathcal{X}}(x_k + \alpha_k d_k)$\;
        }
      \end{algorithm}

def mpg_step(x, grad, alpha=0.01):
    d_k = -np.mean(grad(x), axis=0)

\subsection{Phương pháp tối ưu hóa đa mục tiêu}
\begin{frame}{Nghiên cứu tối ưu hóa đa mục tiêu}{Hệ thống tham chiếu chéo toàn diện và trích dẫn tự động}
  Xét bài toán tối ưu hóa đa mục tiêu với tập ràng buộc $C \subset \mathbb{R}^n$:
  \begin{equation}
    \min_{x \in C} \; F(x) = \big(f_1(x), f_2(x), \dots, f_m(x)\big) \label{eq:mop}
  \end{equation}
  Nghiên cứu của Bello-Cruz và cộng sự\cite{bellocruz2025} giải quyết \Cref{eq:mop} theo tiêu chuẩn \cref{def:pareto}:
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item \textbf{Mô hình \& hình học}: Bài toán \Cref{eq:mop} liên kết với miền giải tích \cref{fig:cauchy} qua công thức tích phân vi phân \eqref{eq:cauchy}.
    \item \textbf{Thuật toán \& cài đặt}: Quy trình giải lặp quán tính\cite{alvarez2024} được đặc tả tại \cref{algo:mpg}, mã nguồn mẫu tại \cref{lst:python} và \cref{lst:cpp}.
    \item \textbf{Hiệu năng thực nghiệm}: Bảng so sánh \cref{tab:exp} khẳng định thuật toán hội tụ về nghiệm Pareto \cref{def:pareto} trên hệ thống chuẩn\cite{utb2026}.
    \item \textbf{Chân trang thông minh}: Cụm trích dẫn tránh logo (\texttt{footerlogo=mark}), tự động rút gọn \texttt{...} trên 1 dòng.
  \end{itemize}
\end{frame}

\subsection{Thuật toán quán tính trong không gian Hilbert}
\begin{frame}{Nhiều trích dẫn trên cùng một slide}{Khóa tối đa 4 dòng chân trang: tự động chia 2 cột và rút gọn \texttt{...} khi có $>4$ mục}
  Các thuật toán giải bài toán tối ưu với định dạng \texttt{tasks} mặc định:
  \begin{tasks}(2)
    \task Phương pháp gradient gần kề\cite{bellocruz2025}
    \task Thuật toán quán tính Hilbert\cite{alvarez2024}
    \task Chuẩn hóa nhận diện Trường Đại học Tây Bắc\cite{utb2026}
    \task Phương pháp đối ngẫu tăng tốc\cite{nesterov2018}
  \end{tasks}
  \vskip 0.5em
  \begin{itemize}
    \item Môi trường \texttt{tasks} mặc định kế thừa dấu đầu dòng hình vuông đặc trưng của Beamer itemize.
    \begin{itemize}[nosep]
      \item Hộp chân trang khóa tối đa \textbf{4 dòng}: tự động rút gọn tên bài báo dài bằng \texttt{...} và chuyển sang \note{bố cục 2 cột}{Tự động kích hoạt khi số trích dẫn và ghi chú trên slide vượt quá 4 dòng} (tối đa 8 mục).
      \item Phân cấp danh sách cấp 3 (dùng \texttt{enumerate} để \note{đánh số La Mã}{Bộ đếm cấp 3 tự động tăng i, ii, iii chính xác} tự động tăng chính xác):
      \begin{enumerate}[label = \roman*)]
        \item Bước kiểm thử thứ nhất (\texttt{i})
        \item Bước kiểm thử kế tiếp tự động tăng đúng số (\texttt{ii})
      \end{enumerate}
    \end{itemize}
  \end{itemize}
\end{frame}
